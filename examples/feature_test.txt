// ============================================================
// 功能测试文件 feature_test.txt
// 每段 print 后面注释了【预期输出】，方便人工核对
// ============================================================

// ---------- 1. 变量独立性（原版 Bug：int a,b 共享对象） ----------
int a, b;
a = 10;
print(a);   // 【10】
print(b);   // 【0】 b 不受 a 影响

// ---------- 2. 算术与优先级 ----------
print(2 + 3 * 4);       // 【14】
print((2 + 3) * 4);     // 【20】
print(17 / 5);          // 【3】  整数除法
print(17 % 5);          // 【2】
print(17.0 / 5);        // 【3.4】
print(2 - 3 - 4);       // 【-5】 左结合
print(-(2 + 3) * 2);    // 【-10】 一元负号
float pi;
pi = 3.14;
print(pi * 2);          // 【6.28】

// ---------- 3. print 直接求值表达式（原版 Bug：输出 12104） ----------
print((1 + 2) * (10 - 4));  // 【18】
print(1, 2, 1 + 2);         // 【1 2 3】 多参数

// ---------- 4. 比较与逻辑（含短路） ----------
print(3 > 2);           // 【true】
print(3 <= 2);          // 【false】
print(5 == 5.0);        // 【true】 int/float 互比
print(5 != 6);          // 【true】
print(true && false);   // 【false】
print(!false || false); // 【true】
print(false && 1 > 2);  // 【false】 短路，右侧不求值

// ---------- 5. 位运算 ----------
print(12 & 10);         // 【8】  1100 & 1010
print(12 | 10);         // 【14】 1100 | 1010
print(~0);              // 【-1】

// ---------- 6. 字符串与字符 ----------
string name;
name = "score: " + 100;
print(name);            // 【score: 100】 字符串拼接数值
char mark;
mark = 'Z';
print(mark);            // 【Z】

// ---------- 7. 数组（原版只有半成品声明） ----------
int fib[8];
fib[0] = 0;
fib[1] = 1;
int k;
k = 2;
while (k < 8) {
    fib[k] = fib[k - 1] + fib[k - 2];
    k = k + 1;
}
print(fib);             // 【[0, 1, 1, 2, 3, 5, 8, 13]】

// ---------- 8. if / else if / else ----------
int score;
score = 82;
if (score >= 90) {
    print("A");
} else if (score >= 80) {
    print("B");         // 【B】
} else if (score >= 60) {
    print("C");
} else {
    print("D");
}

// ---------- 9. while + continue：1~5 奇数和 ----------
int sum, i;
i = 0;
while (i < 5) {
    i = i + 1;
    if (i % 2 == 0) {
        continue;
    }
    sum = sum + i;
}
print(sum);             // 【9】 = 1+3+5

// ---------- 10. 嵌套循环 + break：乘法表按行找第一个 >= 20 的积 ----------
int x, y, found;
found = 0;
x = 1;
while (x <= 9 && found == 0) {
    y = 1;
    while (y <= 9) {
        if (x * y >= 20) {
            print(x, y, x * y);  // 【3 7 21】 第 3 行 3*7=21 先于 4*5=20 出现
            found = 1;
            break;                // 只跳出内层
        }
        y = y + 1;
    }
    x = x + 1;
}

// ---------- 11. 块级作用域：内外同名变量互不影响 ----------
int v;
v = 1;
{
    int v;
    v = 99;
    print(v);           // 【99】
}
print(v);               // 【1】

// ---------- 12. 综合：判断 17 是不是素数 ----------
int n, isPrime, d;
n = 17;
isPrime = 1;
d = 2;
while (d * d <= n) {
    if (n % d == 0) {
        isPrime = 0;
        break;
    }
    d = d + 1;
}
if (isPrime == 1) {
    print(n, "is prime");   // 【17 is prime】
} else {
    print(n, "not prime");
}
